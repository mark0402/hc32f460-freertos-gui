# 第 14 章 综合项目

这一章不做新外设，而是把前面所有知识组合成一个能实际运行的系统。完成之后，你会对"一个嵌入式项目该怎么组织代码"有具体认识，也能亲身体会到裸机主循环架构的局限——这正是下一章引入 RTOS 的动机。

---

## 14.1 项目需求

做一个**环境监测终端**，功能如下：

| 功能 | 实现 | 用到的知识 |
|---|---|---|
| 系统运行在 200MHz | `BSP_CLK_Init` | 第 4 章 |
| 每秒采集一次电压 | Timer0 + ADC | 第 9、11 章 |
| LCD 显示电压/界面 | SPI + DMA | 第 10 章 |
| 串口上报数据 | USART + printf | 第 8 章 |
| 按键切换显示界面 | 外部中断 | 第 5、7 章 |
| LED 指示状态 | GPIO | 第 7 章 |
| 电压过低告警 | 业务逻辑 | — |

---

## 14.2 代码组织

按第 13 章的分层思想，把代码拆成模块：

```
project/
├── main.c              应用层：主循环、状态机、业务逻辑
├── adc_module.c/.h     模块层：ADC 采样与滤波
├── timer_module.c/.h   模块层：Timer0 周期定时
├── key_module.c/.h     模块层：按键中断
├── tftlcd.c/.h         中间件：LCD 显示（开发板提供）
└── bsp/                板级支持（开发板提供）
```

每个模块对外提供简洁的接口，内部封装实现细节。

## 14.3 模块实现

### ADC 模块

```c
/* adc_module.h */
#ifndef ADC_MODULE_H
#define ADC_MODULE_H

void     ADC_Module_Init(void);
uint16_t ADC_ReadOnce(void);
uint16_t ADC_ReadAvg(uint8_t times);
float    ADC_ToVoltage(uint16_t adcVal);

#endif
```

```c
/* adc_module.c */
#include "adc_module.h"
#include "hc32_ddl.h"

#define ADC_UNIT   (M4_ADC1)
#define ADC_PORT   (PortA)
#define ADC_PIN    (Pin00)
#define VREF       (3.3f)
#define ADC_MAX    (4095.0f)

void ADC_Module_Init(void)
{
    stc_adc_init_t  stcAdcInit;
    stc_port_init_t stcPortInit;
    MEM_ZERO_STRUCT(stcAdcInit);
    MEM_ZERO_STRUCT(stcPortInit);

    PWC_Fcg3PeriphClockCmd(PWC_FCG3_PERIPH_ADC1, Enable);   // FCG3

    stcPortInit.enPinMode = Pin_Mode_Ana;                    // 模拟
    stcPortInit.enPullUp  = Disable;
    PORT_Init(ADC_PORT, ADC_PIN, &stcPortInit);

    stcAdcInit.enResolution = AdcResolution_12Bit;
    stcAdcInit.enDataAlign  = AdcDataAlign_Right;
    stcAdcInit.enScanMode   = AdcMode_SAOnce;
    ADC_Init(ADC_UNIT, &stcAdcInit);

    ADC_SetSampleTime(ADC_UNIT, AdcExCh0, 30);
    ADC_Cmd(ADC_UNIT, Enable);
}

uint16_t ADC_ReadOnce(void)
{
    ADC_StartConvert(ADC_UNIT);
    while (Reset == ADC_GetStatus(ADC_UNIT, AdcEoc));
    return ADC_GetValue(ADC_UNIT);
}

uint16_t ADC_ReadAvg(uint8_t times)
{
    uint32_t sum = 0;
    for (uint8_t i = 0; i < times; i++)
    {
        sum += ADC_ReadOnce();
    }
    return (uint16_t)(sum / times);
}

float ADC_ToVoltage(uint16_t adcVal)
{
    return (float)adcVal / ADC_MAX * VREF;
}
```

### 定时器模块

```c
/* timer_module.h */
#ifndef TIMER_MODULE_H
#define TIMER_MODULE_H

void Timer0_Init(void);

#endif
```

```c
/* timer_module.c */
#include "timer_module.h"
#include "hc32_ddl.h"

#define TMR_UNIT     (M4_TMR02)
#define TMR_INI_GCMA (INT_TMR02_GCMA)

/* 由中断置位，主循环消费 */
volatile uint8_t g_sample_flag = 0;
volatile uint32_t g_uptime = 0;

static void Timer0A_CallBack(void)
{
    g_sample_flag = 1;        // 只置标志
    g_uptime++;
}

void Timer0_Init(void)
{
    stc_tim0_base_init_t stcTimerCfg;
    stc_irq_regi_conf_t  stcIrqRegiConf;
    stc_clk_freq_t       stcClkTmp;

    MEM_ZERO_STRUCT(stcTimerCfg);
    MEM_ZERO_STRUCT(stcIrqRegiConf);

    CLK_GetClockFreq(&stcClkTmp);      // 动态读取总线频率

    PWC_Fcg2PeriphClockCmd(PWC_FCG2_PERIPH_TIM02, Enable);   // FCG2

    /* 同步模式，PCLK1，1024 分频，1 秒周期 */
    stcTimerCfg.Tim0_CounterMode     = Tim0_Sync;
    stcTimerCfg.Tim0_SyncClockSource = Tim0_Pclk1;
    stcTimerCfg.Tim0_ClockDivision   = Tim0_ClkDiv1024;
    stcTimerCfg.Tim0_CmpValue        =
        (uint16_t)(stcClkTmp.pclk1Freq / 1024UL / 1UL - 1UL);
    TIMER0_BaseInit(TMR_UNIT, Tim0_ChannelA, &stcTimerCfg);

    TIMER0_IntCmd(TMR_UNIT, Tim0_ChannelA, Enable);

    stcIrqRegiConf.enIRQn      = Int001_IRQn;
    stcIrqRegiConf.enIntSrc    = TMR_INI_GCMA;
    stcIrqRegiConf.pfnCallback = &Timer0A_CallBack;
    enIrqRegistration(&stcIrqRegiConf);
    NVIC_ClearPendingIRQ(stcIrqRegiConf.enIRQn);
    NVIC_SetPriority(stcIrqRegiConf.enIRQn, DDL_IRQ_PRIORITY_DEFAULT);
    NVIC_EnableIRQ(stcIrqRegiConf.enIRQn);

    TIMER0_Cmd(TMR_UNIT, Tim0_ChannelA, Enable);
}
```

### 按键模块

```c
/* key_module.h */
#ifndef KEY_MODULE_H
#define KEY_MODULE_H

void Key_Init(void);

#endif
```

```c
/* key_module.c */
#include "key_module.h"
#include "hc32_ddl.h"

/* 你的板子按键：PC13，上拉，按下为低电平，外部中断通道 13 */
#define KEY_PORT    (PortC)
#define KEY_PIN     (Pin13)
#define KEY_EXINT   (ExtiCh13)              // 外部中断通道
#define KEY_INT_SRC (INT_PORT_EIRQ13)       // 中断源
#define KEY_IRQn    (Int000_IRQn)           // NVIC 槽位（任选一个未占用的槽位即可）

volatile uint8_t g_key_pressed = 0;

static void Key_Callback(void)
{
    if (Set == EXINT_IrqFlgGet(KEY_EXINT))
    {
        g_key_pressed = 1;                   // 只置标志
        EXINT_IrqFlgClr(KEY_EXINT);          // 必须清标志
    }
}

void Key_Init(void)
{
    stc_exint_config_t  stcExtiConfig;
    stc_irq_regi_conf_t stcIrqRegiConf;
    stc_port_init_t     stcPortInit;

    MEM_ZERO_STRUCT(stcExtiConfig);
    MEM_ZERO_STRUCT(stcIrqRegiConf);
    MEM_ZERO_STRUCT(stcPortInit);

    stcExtiConfig.enExitCh   = KEY_EXINT;
    stcExtiConfig.enFilterEn = Enable;              // 硬件消抖
    stcExtiConfig.enFltClk   = Pclk3Div8;
    stcExtiConfig.enExtiLvl  = ExIntFallingEdge;
    EXINT_Init(&stcExtiConfig);

    stcPortInit.enExInt  = Enable;
    stcPortInit.enPullUp = Enable;                 // 上拉：未按下为高电平
    PORT_Init(KEY_PORT, KEY_PIN, &stcPortInit);

    stcIrqRegiConf.enIntSrc    = KEY_INT_SRC;
    stcIrqRegiConf.enIRQn      = KEY_IRQn;
    stcIrqRegiConf.pfnCallback = &Key_Callback;
    enIrqRegistration(&stcIrqRegiConf);
    NVIC_ClearPendingIRQ(stcIrqRegiConf.enIRQn);
    NVIC_SetPriority(stcIrqRegiConf.enIRQn, DDL_IRQ_PRIORITY_DEFAULT);
    NVIC_EnableIRQ(stcIrqRegiConf.enIRQn);
}
```

### 主程序

```c
/* main.c */
#include "hc32_ddl.h"
#include "ev_hc32f460_lqfp100_v1.h"     // 只用其中的 BSP_CLK_Init()
#include "tftlcd.h"
#include "adc_module.h"
#include "timer_module.h"
#include "key_module.h"
#include "stdio.h"

#define VOLTAGE_ALARM_THRESHOLD   (1.0f)     // 告警阈值 1.0V

/* 你的板子用 USART2（PA10/PA15），printf 重定向到它 */
static void Usart2_PortInit(void)
{
    PORT_SetFunc(PortA, Pin10, Func_Usart2_Rx, Disable);
    PORT_SetFunc(PortA, Pin15, Func_Usart2_Tx, Disable);
}

/* LED：PA00/01/02，低电平点亮（见 README 资源表） */
#define LED_RED_PORT    (PortA)
#define LED_RED_PIN     (Pin00)

static uint8_t s_page = 0;                   // 当前界面

/* 界面 0：电压 */
static void ShowPage_Voltage(uint16_t adcVal, float voltage)
{
    char buf[32];
    LCD_Clear(BLACK);
    POINT_COLOR = CYAN;
    LCD_ShowString(0, 0, 240, 16, 16, "Voltage Monitor");

    POINT_COLOR = YELLOW;
    sprintf(buf, "ADC : %4d", adcVal);
    LCD_ShowString(0, 30, 240, 16, 16, buf);

    sprintf(buf, "Volt: %.1f V", voltage);
    LCD_ShowString(0, 50, 240, 16, 16, buf);
}

/* 界面 1：运行时间 */
static void ShowPage_Uptime(void)
{
    char buf[32];
    LCD_Clear(BLACK);
    POINT_COLOR = GREEN;
    LCD_ShowString(0, 0, 240, 16, 16, "System Uptime");

    sprintf(buf, "Uptime: %lu s", g_uptime);
    LCD_ShowString(0, 30, 240, 16, 16, buf);
}

/* 界面 2：系统信息 */
static void ShowPage_Status(void)
{
    LCD_Clear(BLACK);
    POINT_COLOR = MAGENTA;
    LCD_ShowString(0,  0, 240, 16, 16, "System Status");
    POINT_COLOR = WHITE;
    LCD_ShowString(0, 25, 240, 16, 16, "SYSCLK: 200MHz");
    LCD_ShowString(0, 45, 240, 16, 16, "UART  : 115200");
    LCD_ShowString(0, 65, 240, 16, 16, "LCD   : ST7789");
}

int32_t main(void)
{
    uint16_t adcVal = 0;
    float voltage = 0.0f;
    uint8_t lastPage = 0xFF;      // 用 0xFF 强制首次刷新

    /* ===== 初始化（顺序重要） ===== */
    BSP_CLK_Init();               // 1. 时钟
    SystemCoreClockUpdate();      // 2. 更新 SystemCoreClock

    /* 3. LED：PA00/01/02 初始化（低电平点亮） */
    stc_port_init_t stcLedInit;
    MEM_ZERO_STRUCT(stcLedInit);
    stcLedInit.enPinMode = Pin_Mode_Out;
    stcLedInit.enPullUp  = Enable;
    PORT_Init(LED_RED_PORT, LED_RED_PIN, &stcLedInit);   // 红灯 PA00
    PORT_Init(PortA, Pin01, &stcLedInit);
    PORT_Init(PortA, Pin02, &stcLedInit);

    Key_Init();                   // 4. 按键（PC13，含中断）

    /* 5. 串口：USART2，115200（需 ddl_config.h 里 DDL_PRINT_ENABLE = DDL_ON） */
    PWC_Fcg1PeriphClockCmd(PWC_FCG1_PERIPH_USART2, Enable);
    DDL_PrintfInit(M4_USART2, 115200, Usart2_PortInit);

    LCD_Init();                   // 6. LCD（内部含 SPI+DMA）
    LCD_Clear(BLACK);

    ADC_Module_Init();            // 7. ADC
    Timer0_Init();                // 8. 定时器（最后启动，避免中断早于初始化）

    printf("\r\n=== Environment Monitor ===\r\n");
    printf("SYSCLK = %lu Hz\r\n", SystemCoreClock);

    /* ===== 主循环 ===== */
    while (1)
    {
        /* 按键切换界面 */
        if (g_key_pressed)
        {
            g_key_pressed = 0;
            s_page = (s_page + 1) % 3;
            printf("Switch to page %d\r\n", s_page);
        }

        /* 每秒采样 */
        if (g_sample_flag)
        {
            g_sample_flag = 0;

            adcVal = ADC_ReadAvg(16);                    // 16 次平均
            voltage = ADC_ToVoltage(adcVal);

            printf("ADC=%4d  Voltage=%.2fV  Uptime=%lus\r\n",
                   adcVal, voltage, g_uptime);

            PORT_Toggle(PortA, Pin01);                   // 心跳（LED1）

            /* 告警判断：电压过低点亮红灯 PA00 */
            if (voltage < VOLTAGE_ALARM_THRESHOLD)
            {
                PORT_ResetBits(LED_RED_PORT, LED_RED_PIN);
            }
            else
            {
                PORT_SetBits(LED_RED_PORT, LED_RED_PIN);
            }
        }

        /* 界面切换或每秒刷新显示 */
        if ((s_page != lastPage) || g_sample_flag)
        {
            switch (s_page)
            {
                case 0: ShowPage_Voltage(adcVal, voltage); break;
                case 1: ShowPage_Uptime();                 break;
                case 2: ShowPage_Status();                 break;
            }
            lastPage = s_page;
        }

        Ddl_Delay1ms(10);        // 主循环节拍
    }
}
```

## 14.4 设计要点

### 中断只置标志

注意两个中断回调都只做 `g_xxx = 1`，所有业务逻辑在主循环里执行：

```c
static void Timer0A_CallBack(void)
{
    g_sample_flag = 1;        // 只置标志
    g_uptime++;
}
```

这样做的原因（回顾第 5 章）：

- 中断应该尽可能短
- LCD 刷新涉及 SPI + DMA，耗时几毫秒，放在中断里会阻塞其他中断
- 用了 RTOS 之后，中断里更不能调用阻塞 API

### 共享变量用 volatile

```c
volatile uint8_t  g_sample_flag = 0;
volatile uint32_t g_uptime = 0;
volatile uint8_t  g_key_pressed = 0;
```

这三个变量被中断修改、被主循环读取，必须加 `volatile`，否则编译器可能优化掉对内存的重复读取。

### 初始化顺序

```
时钟 → SystemCoreClockUpdate → LED/按键 → 串口 → LCD → ADC → 定时器
```

原则：

**时钟最先**。所有外设的时序都依赖它——串口波特率、定时器周期、SPI 速率。

**定时器最后**。如果定时器先启动而外设还没初始化完，可能在中断里访问未初始化的资源。

**`SystemCoreClockUpdate()` 紧跟时钟配置**。这是最容易忘的一步。

### 用变量保存状态

`lastPage` 记录上次显示的界面，避免每 10ms 重复刷新 LCD：

```c
if ((s_page != lastPage) || g_sample_flag)
```

初值设成 `0xFF`（一个不可能的界面号），保证首次进入时一定刷新。

LCD 刷新是相对耗时的操作（整屏清屏 + 重绘），避免不必要的刷新能显著改善响应速度。

### 模块化的价值

每个外设独立成模块，带来：

- 主程序只有业务逻辑，读起来清晰
- 模块可以单独测试（比如单独测 ADC 模块是否读数正确）
- 下个项目直接复制 `adc_module.c/.h` 复用

## 14.5 这个设计的局限

跑通之后，你可能会注意到几个问题：

### 主循环越来越臃肿

现在只有三个功能，主循环已经包含了按键处理、采样、显示、告警四段逻辑。如果再加一个通信协议解析、一个菜单系统，主循环会变成几百行的 `if` 堆积。

### 时序互相耦合

LCD 刷新（整屏重绘）可能耗时几十毫秒。在刷新期间，即使按了键也不会响应。一个慢操作会拖累整个系统的响应速度。

### 没有真正的优先级

现在"采样"和"显示"是平级的，都在主循环里顺序执行。但实际上采样应该更及时——如果电压突变，晚几十毫秒发现可能是不可接受的。裸机主循环无法表达这种优先级差异。

### 延时浪费 CPU

`Ddl_Delay1ms(10)` 期间 CPU 完全空转。虽然只有 10ms，但如果主循环无事可做，这段时间本可以让芯片进入低功耗。

### 这些问题指向同一个答案

上面四个问题——代码耦合、时序干扰、缺乏优先级、CPU 空转——正是实时操作系统要解决的。

RTOS 把每个功能拆成独立的任务，由调度器根据优先级分配 CPU，任务等待时阻塞而不是空转。下一章展开。

## 14.6 练习

### 练习 1 分阶段实现

不要一次性把整个项目写完，按阶段推进，每阶段都验证：

**阶段 1**：只做 ADC 采样 + 串口打印。确认电压值正确。

**阶段 2**：加 LCD 显示。确认屏幕上显示的数值和串口一致。

**阶段 3**：加定时器，改成每秒自动采样。确认不需要手动触发。

**阶段 4**：加按键切换界面。确认按键响应正常、无抖动误触发。

每完成一个阶段就测试，出问题能快速定位到刚加的部分。这是嵌入式开发的基本方法——一次性写完再调试，往往不知道问题出在哪。

### 练习 2 增加告警功能

在现有基础上增加：

- 电压低于阈值时，红灯闪烁（不是常亮）
- 同时蜂鸣器报警（如果有）
- 串口打印告警信息
- LCD 上显示告警状态

思考：闪烁逻辑怎么实现？如果用 `Ddl_Delay1ms` 在主循环里做，会阻塞其他功能吗？可以用定时器实现非阻塞的闪烁吗？

### 练习 3 数据记录

增加一个环形缓冲区，保存最近 60 次的采样值（1 分钟的历史）。

按键切换到新界面时，在 LCD 上画出这 60 个点的折线图（用 `LCD_Draw_Point` 或画线函数）。

验收：能看到电压随时间变化的曲线。

### 练习 4 性能测量

测量各个操作的耗时：

- 一次 `ADC_ReadAvg(16)` 需要多久
- 一次 LCD 整屏刷新需要多久
- 一次 `printf` 需要多久

方法：用定时器计数值或者翻转一个 GPIO 用示波器测量。

根据测量结果思考：如果主循环里有多个耗时操作，总的循环周期是多少？按键的最坏响应延迟是多少？

### 练习 5 重构与思考

观察你的主循环，回答：

- 如果现在要加一个"每 100ms 检查一次通信超时"的功能，代码要加在哪里？会不会影响现有功能？
- 如果要求"采样必须在 1 秒内完成，不能被显示阻塞"，现有架构能做到吗？
- 如果主循环里某个操作卡死（比如 LCD 通信失败一直等待），整个系统会怎样？

把这些思考记录下来，带着它们进入下一章。

## 14.7 小结

这个项目综合运用了时钟、GPIO、外部中断、USART、SPI+DMA、ADC、定时器等多个外设，并按模块化的方式组织代码。

几个关键实践：中断里只置标志，业务逻辑放主循环；共享变量加 `volatile`；初始化顺序是时钟优先、定时器最后；用状态变量避免不必要的耗时操作。

同时，这个项目也暴露了裸机主循环架构的固有局限：功能耦合、时序互相干扰、无法表达优先级、等待时 CPU 空转。

下一章看 FreeRTOS 如何解决这些问题，以及为什么裸机基础扎实之后再学 RTOS 会轻松很多。
